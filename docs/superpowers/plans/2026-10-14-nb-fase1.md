# nb — Fase 1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Pacchetto Python `nb`: indicizza i `.md` di un knowledge repo git in locale (solo CPU), con embedding committati per hash, ricerca ibrida, hook git di guardia, caricamento guidato e server MCP per Claude Code.

**Architecture:** Moduli piccoli in `src/nb/`. Il core (`fonti`, `chunker`, `vettori`, `indice`, `ricerca`, `guardia`, `caricamento`, `servizi`) non sa nulla di CLI o MCP; `cli.py` e `server.py` sono adattatori sottili. L'indice SQLite è una cache locale derivata da `.md` + `.nb/emb/`.

**Tech Stack:** Python 3.12, uv, fastembed (ONNX), numpy, SQLite FTS5, PyYAML, Typer, MCP Python SDK (FastMCP), pytest.

**Spec:** `docs/specs/2026-10-14-nb-design.md`

## Global Constraints

- Python 3.12 esatto: `requires-python = ">=3.12,<3.13"` (wheel onnxruntime/fastembed).
- Modello default: `sentence-transformers/paraphrase-multilingual-MiniLM-L12-v2`, `dim = 384`, scritto in `.nb/config.toml`.
- Chunk default: `max_parole = 100`, `overlap = 20` (il modello tronca a 128 token).
- Vettori: `.nb/emb/<hash[:2]>/<hash>.bin`, float16 little-endian, 768 byte; `hash = sha256(model_id + "\n" + testo_embed)`.
- `.nb/index.db*` sempre in `.gitignore`, mai committato.
- Solo `.md` indicizzati; binari e file > `max_file_mb` (default 5) bloccati dagli hook.
- Il percorso del file NON entra nel testo embeddato; i titoli interni sì.
- Nessun file condiviso mutabile generato dal sistema (niente manifest).
- Ogni messaggio verso l'utente è in italiano.
- git invocato sempre con `GIT_LITERAL_PATHSPECS=1`, `GIT_TERMINAL_PROMPT=0` e output `-z` per i nomi file.
- Test senza rete né modello: variabile `NB_EMBEDDER=fake` (autouse in `conftest.py`). Il test sul modello reale ha marker `modello`, quello di prestazioni `lento`.
- Push solo dopo conferma esplicita dell'utente (docstring del tool + skill).

## Review Focus

1. Nomi file con spazi e lettere accentate (`riunione è qui.md`) → guardia e staging li trattano come gli altri. Test: Task 8 `test_nome_con_accenti_e_spazi`, `test_in_staging_con_accenti`.
2. Query con sintassi FTS5 (`"`, `-`, `*`, `:`, parentesi, `AND OR NOT`, stringa vuota) → nessuna eccezione. Test: Task 7 `test_query_con_caratteri_speciali`.

5. Remoto irraggiungibile / branch senza upstream → avviso, nessun blocco. Test: Task 8 `test_fetch_fallito_avvisa`, `test_senza_upstream`.

```python
"""Configurazione del knowledge repo (.nb/config.toml)."""
from __future__ import annotations

import tomllib
from dataclasses import dataclass
from pathlib import Path

NB_DIR = ".nb"
MODELLO_DEFAULT = "sentence-transformers/paraphrase-multilingual-MiniLM-L12-v2"
TIPI = ("analisi", "documentazione", "trascrizione", "altro")
CARTELLA_TIPO = {"analisi": "analisi", "documentazione": "documentazione", "call": "trascrizione"}

CONFIG_DEFAULT = f"""\
[embedding]
model_id = "{MODELLO_DEFAULT}"
dim = 384

[chunk]
max_parole = 100
overlap = 20

[regole]
max_file_mb = 5
"""


class ConfigError(RuntimeError):
    pass


@dataclass(frozen=True)
class Config:
    root: Path
    model_id: str = MODELLO_DEFAULT
    dim: int = 384
    max_parole: int = 100
    overlap: int = 20
    max_file_mb: float = 5.0

    @property
    def nb_dir(self) -> Path:
        return self.root / NB_DIR

    @property
    def emb_dir(self) -> Path:
        return self.nb_dir / "emb"

    @property
    def index_path(self) -> Path:
        return self.nb_dir / "index.db"


def file_config(root: Path) -> Path:
    return root / NB_DIR / "config.toml"


def trova_root(inizio: Path) -> Path:
    inizio = inizio.resolve()
    for p in (inizio, *inizio.parents):
        if file_config(p).is_file():
            return p
    raise ConfigError("nessun knowledge repo nb qui (manca .nb/config.toml): esegui `nb init`")


def carica(root: Path) -> Config:
    try:
        dati = tomllib.loads(file_config(root).read_text(encoding="utf-8"))
    except FileNotFoundError:
        raise ConfigError(f"manca {file_config(root)}: esegui `nb init`") from None
    except tomllib.TOMLDecodeError as e:
        raise ConfigError(f".nb/config.toml non valido: {e}") from None
    emb, chunk, regole = dati.get("embedding", {}), dati.get("chunk", {}), dati.get("regole", {})
    config = Config(
        root=root.resolve(),
        model_id=str(emb.get("model_id", MODELLO_DEFAULT)),
        dim=int(emb.get("dim", 384)),
        max_parole=int(chunk.get("max_parole", 100)),
        overlap=int(chunk.get("overlap", 20)),
        max_file_mb=float(regole.get("max_file_mb", 5)),
    )
    if config.overlap >= config.max_parole:
        raise ConfigError("chunk.overlap deve essere minore di chunk.max_parole")
    return config


def scrivi_default(root: Path) -> bool:
    """Scrive la config di default se manca. True se l'ha scritta."""
    f = file_config(root)
    if f.exists():
        return False
    f.parent.mkdir(parents=True, exist_ok=True)
    f.write_text(CONFIG_DEFAULT, encoding="utf-8")
    return True
```

```python
"""Scoperta dei .md indicizzabili e dei loro metadati (frontmatter + percorso)."""
from __future__ import annotations

import re
from dataclasses import dataclass
from pathlib import Path, PurePosixPath

import yaml

from nb.config import CARTELLA_TIPO, TIPI

ESCLUSI_RADICE = {"README.md", "CLAUDE.md"}
RE_DATA = re.compile(r"^(\d{4}-\d{2}-\d{2})-")


@dataclass(frozen=True)
class Metadati:
    tipo: str = "altro"
    progetto: str | None = None
    data: str | None = None
    partecipanti: tuple[str, ...] = ()


@dataclass(frozen=True)
class Fonte:
    percorso: str
    corpo: str
    meta: Metadati
    riga_corpo: int  # numero (1-based) della prima riga del corpo nel file
    avvisi: tuple[str, ...] = ()


def indicizzabile(percorso: str) -> bool:
    parti = PurePosixPath(percorso).parts
    if not parti or not percorso.endswith(".md"):
        return False
    if any(p.startswith(".") for p in parti[:-1]):
        return False
    return not (len(parti) == 1 and parti[0] in ESCLUSI_RADICE)


def elenca_md(root: Path) -> list[str]:
    trovati = []
    for p in root.rglob("*.md"):
        rel = p.relative_to(root).as_posix()
        if p.is_file() and indicizzabile(rel):
            trovati.append(rel)
    return sorted(trovati)


def separa_frontmatter(testo: str) -> tuple[dict | None, str, int, str | None]:
    """Restituisce (frontmatter, corpo, riga_corpo, errore)."""
    testo = testo.removeprefix("﻿")
    righe = testo.split("\n")
    if righe[0].strip() != "---":
        return None, testo, 1, None
    for i in range(1, len(righe)):
        if righe[i].strip() in ("---", "..."):
            grezzo = "\n".join(righe[1:i])
            corpo = "\n".join(righe[i + 1:])
            try:
                dati = yaml.safe_load(grezzo) or {}
            except yaml.YAMLError as e:
                return None, corpo, i + 2, f"frontmatter YAML non valido ({e.__class__.__name__})"
            if not isinstance(dati, dict):
                return None, corpo, i + 2, "frontmatter non è una mappa chiave: valore"
            return dati, corpo, i + 2, None
    return None, testo, 1, "frontmatter aperto con --- ma mai chiuso"


def meta_da_percorso(percorso: str) -> Metadati:
    parti = PurePosixPath(percorso).parts
    in_progetti = parti[0] == "progetti"
    progetto = parti[1] if in_progetti and len(parti) >= 3 else None
    tipo = CARTELLA_TIPO.get(parti[2], "altro") if in_progetti and len(parti) >= 4 else "altro"
    m = RE_DATA.match(parti[-1])
    return Metadati(tipo=tipo, progetto=progetto, data=m.group(1) if m else None)


def _unisci(base: Metadati, fm: dict) -> tuple[Metadati, list[str]]:
    avvisi = []
    tipo = fm.get("tipo", base.tipo)
    if tipo not in TIPI:
        avvisi.append(f"tipo '{tipo}' non ammesso (ammessi: {', '.join(TIPI)})")
        tipo = base.tipo
    progetto = fm.get("progetto", base.progetto)
    data = fm.get("data", base.data)
    partecipanti = fm.get("partecipanti", [])
    if isinstance(partecipanti, str):
        partecipanti = [partecipanti]
    if not isinstance(partecipanti, list):
        avvisi.append("partecipanti deve essere una lista")
        partecipanti = []
    meta = Metadati(
        tipo=tipo,
        progetto=None if progetto is None else str(progetto),
        data=None if data is None else str(data),
        partecipanti=tuple(str(p) for p in partecipanti),
    )
    return meta, avvisi


def analizza(percorso: str, testo: str) -> Fonte:
    fm, corpo, riga_corpo, errore = separa_frontmatter(testo)
    meta = meta_da_percorso(percorso)
    avvisi = [errore] if errore else []
    if fm:
        meta, extra = _unisci(meta, fm)
        avvisi += extra
    return Fonte(percorso, corpo, meta, riga_corpo, tuple(avvisi))


def leggi_fonte(root: Path, percorso: str) -> Fonte:
    return analizza(percorso, (root / percorso).read_text(encoding="utf-8", errors="replace"))
```

```python
"""Regole di validazione condivise da hook git e caricamento."""
from __future__ import annotations

import re
from dataclasses import dataclass
from pathlib import PurePosixPath

from nb.config import CARTELLA_TIPO, TIPI
from nb.fonti import separa_frontmatter

RE_CALL = re.compile(r"^\d{4}-\d{2}-\d{2}-[a-z0-9]+(?:-[a-z0-9]+)*\.md$")
RE_SLUG = re.compile(r"^[a-z0-9]+(?:-[a-z0-9]+)*\.md$")
RE_CONFLITTO = re.compile(r"^(?:<{7}|>{7})(?: |$)", re.M)
SEGRETI_CERTI = (
    ("chiave privata", re.compile(r"-----BEGIN [A-Z ]*PRIVATE KEY-----")),
    ("AWS access key", re.compile(r"\bAKIA[0-9A-Z]{16}\b")),
    ("token GitHub", re.compile(r"\bgh[pousr]_[A-Za-z0-9]{36,}\b")),
    ("token GitLab", re.compile(r"\bglpat-[A-Za-z0-9_-]{20,}")),
    ("chiave Anthropic", re.compile(r"\bsk-ant-[A-Za-z0-9_-]{20,}")),
    ("token Slack", re.compile(r"\bxox[abprs]-[A-Za-z0-9-]{10,}")),
)
RE_SEGRETO_PROBABILE = re.compile(r"(?i)\b(?:password|passwd|pwd|secret|token|api[_-]?key)\s*[:=]\s*\S+")
ESTENSIONI_BINARIE = frozenset({
    ".pdf", ".doc", ".docx", ".xls", ".xlsx", ".ppt", ".pptx", ".odt",
    ".mp3", ".wav", ".m4a", ".ogg", ".flac", ".mp4", ".mov", ".mkv", ".webm", ".avi",
    ".png", ".jpg", ".jpeg", ".gif", ".heic", ".zip", ".7z", ".rar", ".tar", ".gz",
})
ESENTI = frozenset({"README.md", "CLAUDE.md", ".gitignore", ".mcp.json"})
PREFISSI_ESENTI = (".nb/", ".githooks/", ".claude/")


@dataclass(frozen=True)
class Esito:
    livello: str  # "blocco" | "avviso"
    percorso: str
    messaggio: str

    def __str__(self) -> str:
        simbolo = "✋" if self.livello == "blocco" else "⚠"
        return f"{simbolo} {self.percorso}: {self.messaggio}"


def bloccanti(esiti: list[Esito]) -> list[Esito]:
    return [e for e in esiti if e.livello == "blocco"]


def _esente(percorso: str) -> bool:
    return percorso in ESENTI or percorso.startswith(PREFISSI_ESENTI)


def controlla_percorso(percorso: str) -> list[Esito]:
    if _esente(percorso) or not percorso.endswith(".md"):
        return []
    parti = PurePosixPath(percorso).parts
    nome = parti[-1]
    in_progetto = len(parti) >= 4 and parti[0] == "progetti" and parti[2] in CARTELLA_TIPO
    in_generale = len(parti) >= 2 and parti[0] == "generale"
    esiti = []
    if not (in_progetto or in_generale):
        esiti.append(Esito("avviso", percorso,
            "fuori dalla struttura prevista: progetti/<progetto>/{analisi,documentazione,call}/ oppure generale/"))
    if in_progetto and parti[2] == "call":
        if not RE_CALL.match(nome):
            esiti.append(Esito("blocco", percorso,
                "le trascrizioni si chiamano AAAA-MM-GG-titolo.md (minuscolo, trattini)"))
    elif not RE_SLUG.match(nome):
        esiti.append(Esito("avviso", percorso, "nome consigliato: minuscolo-con-trattini.md"))
    return esiti


def controlla_md(percorso: str, testo: str) -> list[Esito]:
    esiti = []
    fm, _, _, errore = separa_frontmatter(testo)
    if errore:
        esiti.append(Esito("blocco", percorso, errore))
    elif fm and "tipo" in fm and fm["tipo"] not in TIPI:
        esiti.append(Esito("blocco", percorso, f"tipo '{fm['tipo']}' non ammesso (ammessi: {', '.join(TIPI)})"))
    if RE_CONFLITTO.search(testo):
        esiti.append(Esito("blocco", percorso, "marcatori di conflitto git (<<<<<<< / >>>>>>>) nel testo"))
    for nome, rx in SEGRETI_CERTI:
        if rx.search(testo):
            esiti.append(Esito("blocco", percorso, f"possibile segreto ({nome}): rimuovilo prima di committare"))
    if RE_SEGRETO_PROBABILE.search(testo):
        esiti.append(Esito("avviso", percorso, "possibile credenziale nel testo (password/token): controlla"))
    return esiti


def controlla_file(percorso: str, dati: bytes, max_file_mb: float) -> list[Esito]:
    if percorso.startswith(".nb/"):
        return []
    if PurePosixPath(percorso).suffix.lower() in ESTENSIONI_BINARIE:
        return [Esito("blocco", percorso, "file binario: nel repo vanno solo .md (gli originali restano fuori)")]
    esiti = []
    if len(dati) > max_file_mb * 1024 * 1024:
        esiti.append(Esito("blocco", percorso, f"file oltre {max_file_mb:g} MB"))
    if percorso.endswith(".md"):
        esiti += controlla_percorso(percorso)
        esiti += controlla_md(percorso, dati.decode("utf-8", errors="replace"))
    elif not _esente(percorso):
        esiti.append(Esito("avviso", percorso, "non è un .md: non verrà indicizzato"))
    return esiti
```

```python
"""Spezza una Fonte in chunk: per titoli (documenti) o per battute (trascrizioni)."""
from __future__ import annotations

import re
from dataclasses import dataclass

from nb.config import Config
from nb.fonti import Fonte, analizza

RE_TITOLO = re.compile(r"^(#{1,3})\s+(.+?)\s*#*\s*$")
RE_FENCE = re.compile(r"^\s*(```|~~~)")
RE_SPEAKER = re.compile(
    r"^\s*(?:"
    r"\*\*(?P<a>[^*:\n]{1,60})\*\*\s*(?:\[[\d:]{4,8}\])?\s*:"
    r"|\*\*(?P<b>[^*:\n]{1,60}?)\s*(?:\[[\d:]{4,8}\])?\s*:\*\*"
    r"|(?P<c>[A-ZÀ-Ý][\w'.-]*(?: [A-ZÀ-Ý][\w'.-]*){0,3})\s*(?:\[[\d:]{4,8}\])?\s*:(?:\s|$)"
    r")"
)

Riga = tuple[int, str]  # (numero riga nel file, testo)


@dataclass(frozen=True)
class Chunk:
    percorso: str
    titoli: tuple[str, ...]
    testo: str
    riga_inizio: int
    riga_fine: int
    speaker: tuple[str, ...] = ()

    @property
    def testo_embed(self) -> str:
        if not self.titoli:
            return self.testo
        return " › ".join(self.titoli) + "\n" + self.testo


def _parole(s: str) -> int:
    return len(s.split())


def speaker_di(riga: str) -> str | None:
    m = RE_SPEAKER.match(riga)
    if not m:
        return None
    return next(v for v in (m.group("a"), m.group("b"), m.group("c")) if v).strip()


def _impacchetta(unita: list[list[Riga]], max_parole: int, overlap: int) -> list[list[Riga]]:
    """Raggruppa unità atomiche (paragrafi o battute) in finestre di circa max_parole."""
    piatte: list[list[Riga]] = []
    for u in unita:
        if sum(_parole(t) for _, t in u) > max_parole:
            piatte.extend([r] for r in u)
        else:
            piatte.append(u)
    finestre: list[list[Riga]] = []
    corrente: list[Riga] = []
    n = 0
    for u in piatte:
        pu = sum(_parole(t) for _, t in u)
        if corrente and n + pu > max_parole:
            finestre.append(corrente)
            coda: list[Riga] = []
            m = 0
            for r in reversed(corrente[1:]) if overlap else ():
                if m + _parole(r[1]) > overlap:
                    break
                coda.insert(0, r)
                m += _parole(r[1])
            corrente, n = coda, m
        corrente.extend(u)
        n += pu
    if corrente:
        finestre.append(corrente)
    return finestre


def _crea(percorso: str, titoli: tuple[str, ...], finestra: list[Riga], speaker: tuple[str, ...]) -> Chunk | None:
    righe = [r for r in finestra]
    while righe and not righe[0][1].strip():
        righe.pop(0)
    while righe and not righe[-1][1].strip():
        righe.pop()
    if not righe or not any(_parole(t) for _, t in righe):
        return None
    testo = "\n".join(t for _, t in righe)
    return Chunk(percorso, titoli, testo, righe[0][0], righe[-1][0], speaker)


def _paragrafi(righe: list[Riga]) -> list[list[Riga]]:
    out, par = [], []
    for r in righe:
        par.append(r)
        if not r[1].strip():
            out.append(par)
            par = []
    if par:
        out.append(par)
    return out


def _spezza_documento(fonte: Fonte, righe: list[Riga], max_parole: int, overlap: int) -> list[Chunk]:
    sezioni: list[tuple[tuple[str, ...], list[Riga]]] = []
    pila: list[tuple[int, str]] = []
    titoli: tuple[str, ...] = ()
    corrente: list[Riga] = []
    in_codice = False
    for n, riga in righe:
        if RE_FENCE.match(riga):
            in_codice = not in_codice
        m = None if in_codice else RE_TITOLO.match(riga)
        if m:
            sezioni.append((titoli, corrente))
            livello = len(m.group(1))
            pila = [(lv, t) for lv, t in pila if lv < livello] + [(livello, m.group(2))]
            titoli = tuple(t for _, t in pila)
            corrente = []
        else:
            corrente.append((n, riga))
    sezioni.append((titoli, corrente))
    chunks = []
    for tit, righe_sezione in sezioni:
        for finestra in _impacchetta(_paragrafi(righe_sezione), max_parole, overlap):
            c = _crea(fonte.percorso, tit, finestra, ())
            if c:
                chunks.append(c)
    return chunks


def _spezza_trascrizione(fonte: Fonte, righe: list[Riga], max_parole: int) -> list[Chunk]:
    battute: list[list[Riga]] = []
    corrente: list[Riga] = []
    for r in righe:
        if speaker_di(r[1]) and corrente:
            battute.append(corrente)
            corrente = []
        corrente.append(r)
    if corrente:
        battute.append(corrente)
    chunks = []
    for finestra in _impacchetta(battute, max_parole, 0):
        speaker = tuple(dict.fromkeys(s for s in (speaker_di(t) for _, t in finestra) if s))
        c = _crea(fonte.percorso, (), finestra, speaker)
        if c:
            chunks.append(c)
    return chunks


def spezza(fonte: Fonte, max_parole: int, overlap: int) -> list[Chunk]:
    if not fonte.corpo.strip():
        return []
    righe = [(fonte.riga_corpo + i, t) for i, t in enumerate(fonte.corpo.split("\n"))]
    if fonte.meta.tipo == "trascrizione":
        return _spezza_trascrizione(fonte, righe, max_parole)
    return _spezza_documento(fonte, righe, max_parole, overlap)


def chunk_di(config: Config, percorso: str, testo: str) -> tuple[Fonte, list[Chunk]]:
    fonte = analizza(percorso, testo)
    return fonte, spezza(fonte, config.max_parole, config.overlap)
```

import os
import re
from pathlib import Path
from typing import Protocol

import numpy as np

from nb.config import Config

CACHE_MODELLI = Path.home() / ".cache" / "nb" / "fastembed"


class Embedder(Protocol):
    model_id: str
    dim: int

    def embed(self, testi: list[str]) -> np.ndarray: ...

    def embed_query(self, testo: str) -> np.ndarray: ...


def _normalizza(m: np.ndarray) -> np.ndarray:
    norme = np.linalg.norm(m, axis=1, keepdims=True)
    norme[norme == 0] = 1
    return (m / norme).astype(np.float32)


class FastEmbedder:
    def __init__(self, model_id: str, dim: int):
        self.model_id = model_id
        self.dim = dim
        self._modello = None

    def _m(self):
        if self._modello is None:
            from fastembed import TextEmbedding

            self._modello = TextEmbedding(model_name=self.model_id, cache_dir=str(CACHE_MODELLI))
        return self._modello

    def embed(self, testi: list[str]) -> np.ndarray:
        if not testi:
            return np.zeros((0, self.dim), np.float32)
        return _normalizza(np.array(list(self._m().embed(testi, batch_size=64)), dtype=np.float32))

    def embed_query(self, testo: str) -> np.ndarray:
        return self.embed([testo])[0]


class FakeEmbedder:
    """Deterministico e senza download: bag-of-words con hashing. Solo per i test."""

    def __init__(self, model_id: str = "fake", dim: int = 384):
        self.model_id = model_id
        self.dim = dim

    def embed(self, testi: list[str]) -> np.ndarray:
        m = np.zeros((len(testi), self.dim), np.float32)
        for i, t in enumerate(testi):
            for parola in re.findall(r"\w+", t.lower()):
                h = int.from_bytes(hashlib.blake2b(parola.encode(), digest_size=4).digest(), "little")
                m[i, h % self.dim] += 1.0
        return _normalizza(m)

    def embed_query(self, testo: str) -> np.ndarray:
        return self.embed([testo])[0]


def crea(config: Config) -> Embedder:
    if os.environ.get("NB_EMBEDDER") == "fake":
        return FakeEmbedder(config.model_id, config.dim)
    return FastEmbedder(config.model_id, config.dim)

```python
"""Vettori committati: un file immutabile per chunk, nominato con l'hash del contenuto."""
from __future__ import annotations

import hashlib
from pathlib import Path

import numpy as np

from nb.embedder import Embedder


def chiave(model_id: str, testo: str) -> str:
    return hashlib.sha256(f"{model_id}\n{testo}".encode()).hexdigest()


def percorso(emb_dir: Path, h: str) -> Path:
    return emb_dir / h[:2] / f"{h}.bin"


def percorso_rel(h: str) -> str:
    return f".nb/emb/{h[:2]}/{h}.bin"


def leggi(emb_dir: Path, h: str, dim: int) -> np.ndarray | None:
    try:
        dati = percorso(emb_dir, h).read_bytes()
    except FileNotFoundError:
        return None
    if len(dati) != dim * 2:
        return None
    return np.frombuffer(dati, dtype="<f2").astype(np.float32)


def scrivi(emb_dir: Path, h: str, vettore: np.ndarray) -> Path:
    p = percorso(emb_dir, h)
    p.parent.mkdir(parents=True, exist_ok=True)
    tmp = p.with_suffix(".tmp")
    tmp.write_bytes(np.asarray(vettore, dtype="<f2").tobytes())
    tmp.replace(p)
    return p


def assicura(emb_dir: Path, embedder: Embedder, testi: dict[str, str]) -> tuple[dict[str, np.ndarray], list[Path]]:
    """Carica i vettori esistenti, calcola e scrive solo i mancanti."""
    trovati: dict[str, np.ndarray] = {}
    mancanti: list[str] = []
    for h in testi:
        v = leggi(emb_dir, h, embedder.dim)
        if v is None:
            mancanti.append(h)
        else:
            trovati[h] = v
    nuovi: list[Path] = []
    if mancanti:
        calcolati = embedder.embed([testi[h] for h in mancanti])
        for h, v in zip(mancanti, calcolati):
            nuovi.append(scrivi(emb_dir, h, v))
            trovati[h] = v.astype("<f2").astype(np.float32)
    return trovati, nuovi


def tutte(emb_dir: Path) -> set[str]:
    return {p.stem for p in emb_dir.glob("*/*.bin")}


def gc(emb_dir: Path, vivi: set[str]) -> list[Path]:
    rimossi = []
    for p in sorted(emb_dir.glob("*/*.bin")):
        if p.stem not in vivi:
            p.unlink()
            rimossi.append(p)
    for d in emb_dir.glob("*"):
        if d.is_dir() and not any(d.iterdir()):
            d.rmdir()
    return rimossi
```

```python
"""Indice locale (SQLite + FTS5): cache derivata da .md e .nb/emb, mai committata."""
from __future__ import annotations

import hashlib
import json
import sqlite3
from dataclasses import dataclass, field
from pathlib import Path

import numpy as np

from nb.chunker import Chunk, chunk_di
from nb.config import Config
from nb.embedder import Embedder
from nb.fonti import Fonte, elenca_md
from nb.vettori import assicura, chiave

SCHEMA = "1"
DDL = """
CREATE TABLE meta(k TEXT PRIMARY KEY, v TEXT);
CREATE TABLE file(percorso TEXT PRIMARY KEY, mtime_ns INTEGER, sha TEXT, tipo TEXT,
                  progetto TEXT, data TEXT, partecipanti TEXT, avvisi TEXT);
CREATE TABLE chunk(id INTEGER PRIMARY KEY, percorso TEXT NOT NULL, titoli TEXT, testo TEXT,
                   riga_inizio INTEGER, riga_fine INTEGER, speaker TEXT, hash TEXT, vettore BLOB);
CREATE INDEX chunk_percorso ON chunk(percorso);
CREATE VIRTUAL TABLE chunk_fts USING fts5(testo, titoli, tokenize='unicode61 remove_diacritics 2');
"""


@dataclass
class ReportSync:
    aggiunti: list[str] = field(default_factory=list)
    modificati: list[str] = field(default_factory=list)
    rimossi: list[str] = field(default_factory=list)
    vettori_nuovi: list[Path] = field(default_factory=list)

    @property
    def cambiato(self) -> bool:
        return bool(self.aggiunti or self.modificati or self.rimossi)


@dataclass(frozen=True)
class Matrice:
    ids: np.ndarray
    vettori: np.ndarray
    percorsi: np.ndarray


class Indice:
    def __init__(self, config: Config, embedder: Embedder):
        self.config = config
        self.embedder = embedder
        self.db = self._apri()
        self._matrice: Matrice | None = None

    def _connetti(self) -> sqlite3.Connection:
        db = sqlite3.connect(self.config.index_path, timeout=10, check_same_thread=False)
        db.execute("PRAGMA journal_mode=WAL")
        return db

    def _apri(self) -> sqlite3.Connection:
        self.config.nb_dir.mkdir(parents=True, exist_ok=True)
        db = None
        try:
            db = self._connetti()
            meta = dict(db.execute("SELECT k, v FROM meta"))
            if meta.get("schema") == SCHEMA and meta.get("model_id") == self.embedder.model_id:
                return db
        except sqlite3.DatabaseError:
            pass
        if db is not None:
            db.close()
        for suffisso in ("", "-wal", "-shm"):
            Path(f"{self.config.index_path}{suffisso}").unlink(missing_ok=True)
        db = self._connetti()
        db.executescript(DDL)
        db.executemany("INSERT INTO meta VALUES (?, ?)", [("schema", SCHEMA), ("model_id", self.embedder.model_id)])
        db.commit()
        return db

    def sync(self) -> ReportSync:
        root = self.config.root
        rep = ReportSync()
        noti = {p: (mt, sha) for p, mt, sha in self.db.execute("SELECT percorso, mtime_ns, sha FROM file")}
        su_disco = elenca_md(root)
        da_elaborare = []
        for p in su_disco:
            mtime = (root / p).stat().st_mtime_ns
            if p in noti and noti[p][0] == mtime:
                continue
            dati = (root / p).read_bytes()
            sha = hashlib.sha256(dati).hexdigest()
            if p in noti and noti[p][1] == sha:
                self.db.execute("UPDATE file SET mtime_ns = ? WHERE percorso = ?", (mtime, p))
                continue
            (rep.modificati if p in noti else rep.aggiunti).append(p)
            da_elaborare.append((p, dati.decode("utf-8", errors="replace"), sha, mtime))
        rep.rimossi = sorted(set(noti) - set(su_disco))
        for p in rep.rimossi:
            self._rimuovi(p)
        elaborati, testi = [], {}
        for p, testo, sha, mtime in da_elaborare:
            fonte, chunks = chunk_di(self.config, p, testo)
            hashes = [chiave(self.embedder.model_id, c.testo_embed) for c in chunks]
            testi.update(zip(hashes, (c.testo_embed for c in chunks)))
            elaborati.append((p, sha, mtime, fonte, chunks, hashes))
        vettori, rep.vettori_nuovi = assicura(self.config.emb_dir, self.embedder, testi)
        for p, sha, mtime, fonte, chunks, hashes in elaborati:
            self._rimuovi(p)
            self._inserisci(p, sha, mtime, fonte, chunks, hashes, vettori)
        self.db.commit()
        if rep.cambiato:
            self._matrice = None
        return rep

    def _rimuovi(self, p: str) -> None:
        self.db.execute("DELETE FROM chunk_fts WHERE rowid IN (SELECT id FROM chunk WHERE percorso = ?)", (p,))
        self.db.execute("DELETE FROM chunk WHERE percorso = ?", (p,))
        self.db.execute("DELETE FROM file WHERE percorso = ?", (p,))

    def _inserisci(self, p: str, sha: str, mtime: int, fonte: Fonte, chunks: list[Chunk],
                   hashes: list[str], vettori: dict[str, np.ndarray]) -> None:
        m = fonte.meta
        self.db.execute(
            "INSERT INTO file VALUES (?, ?, ?, ?, ?, ?, ?, ?)",
            (p, mtime, sha, m.tipo, m.progetto, m.data,
             json.dumps(m.partecipanti, ensure_ascii=False), json.dumps(fonte.avvisi, ensure_ascii=False)),
        )
        for c, h in zip(chunks, hashes):
            cur = self.db.execute(
                "INSERT INTO chunk (percorso, titoli, testo, riga_inizio, riga_fine, speaker, hash, vettore) "
                "VALUES (?, ?, ?, ?, ?, ?, ?, ?)",
                (p, json.dumps(c.titoli, ensure_ascii=False), c.testo, c.riga_inizio, c.riga_fine,
                 json.dumps(c.speaker, ensure_ascii=False), h, vettori[h].astype("<f4").tobytes()),
            )
            self.db.execute("INSERT INTO chunk_fts (rowid, testo, titoli) VALUES (?, ?, ?)",
                            (cur.lastrowid, c.testo, " ".join(c.titoli)))

    def matrice(self) -> Matrice:
        if self._matrice is None:
            righe = self.db.execute("SELECT id, percorso, vettore FROM chunk ORDER BY id").fetchall()
            self._matrice = Matrice(
                ids=np.array([r[0] for r in righe], dtype=np.int64),
                vettori=np.frombuffer(b"".join(r[2] for r in righe), dtype="<f4").reshape(len(righe), self.embedder.dim),
                percorsi=np.array([r[1] for r in righe], dtype=object),
            )
        return self._matrice

    def avvisi(self) -> dict[str, list[str]]:
        return {p: json.loads(a) for p, a in
                self.db.execute("SELECT percorso, avvisi FROM file WHERE avvisi != '[]' ORDER BY percorso")}

    def hash_vivi(self) -> set[str]:
        return {h for (h,) in self.db.execute("SELECT DISTINCT hash FROM chunk")}

    def close(self) -> None:
        self.db.close()
```

```python
"""Ricerca ibrida (vettoriale + BM25, fusione RRF) e documenti simili."""
from __future__ import annotations

import json
import re
from dataclasses import asdict, dataclass

import numpy as np

from nb.chunker import chunk_di
from nb.indice import Indice

RRF_K = 60
CANDIDATI = 50
SOGLIA_COPERTURA = 0.9


@dataclass(frozen=True)
class Filtri:
    progetto: str | None = None
    tipo: str | None = None
    da: str | None = None
    a: str | None = None


@dataclass(frozen=True)
class Risultato:
    percorso: str
    titoli: tuple[str, ...]
    testo: str
    riga_inizio: int
    riga_fine: int
    speaker: tuple[str, ...]
    tipo: str
    progetto: str | None
    data: str | None
    punteggio: float

    @property
    def citazione(self) -> str:
        return " › ".join((self.percorso, *self.titoli))

    def to_dict(self) -> dict:
        return {**asdict(self), "citazione": self.citazione}


@dataclass(frozen=True)
class Simile:
    percorso: str
    similarita: float
    copertura: float


def _condizioni(f: Filtri) -> tuple[str, list[str]]:
    cond, par = [], []
    for colonna, op, valore in (("progetto", "=", f.progetto), ("tipo", "=", f.tipo),
                                ("data", ">=", f.da), ("data", "<=", f.a)):
        if valore:
            cond.append(f"f.{colonna} {op} ?")
            par.append(valore)
    return " AND ".join(cond), par


def _query_fts(query: str) -> str:
    return " OR ".join(f'"{p}"' for p in re.findall(r"\w+", query.lower()))


def _vettoriale(indice: Indice, query: str, ammessi: set[int] | None) -> list[int]:
    m = indice.matrice()
    if not len(m.ids):
        return []
    sim = m.vettori @ indice.embedder.embed_query(query)
    if ammessi is not None:
        sim = np.where(np.isin(m.ids, list(ammessi)), sim, -np.inf)
    n = min(CANDIDATI, len(m.ids))
    top = np.argpartition(-sim, n - 1)[:n]
    top = top[np.argsort(-sim[top])]
    return [int(m.ids[i]) for i in top if np.isfinite(sim[i])]


def _bm25(indice: Indice, query: str, cond: str, par: list[str]) -> list[int]:
    fts = _query_fts(query)
    if not fts:
        return []
    sql = ("SELECT chunk_fts.rowid FROM chunk_fts JOIN chunk c ON c.id = chunk_fts.rowid "
           "JOIN file f ON f.percorso = c.percorso WHERE chunk_fts MATCH ?"
           + (f" AND {cond}" if cond else "") + " ORDER BY bm25(chunk_fts) LIMIT ?")
    return [r[0] for r in indice.db.execute(sql, [fts, *par, CANDIDATI])]


def _carica(indice: Indice, cid: int, punteggio: float) -> Risultato:
    r = indice.db.execute(
        "SELECT c.percorso, c.titoli, c.testo, c.riga_inizio, c.riga_fine, c.speaker, f.tipo, f.progetto, f.data "
        "FROM chunk c JOIN file f ON f.percorso = c.percorso WHERE c.id = ?", (cid,)).fetchone()
    return Risultato(r[0], tuple(json.loads(r[1])), r[2], r[3], r[4], tuple(json.loads(r[5])),
                     r[6], r[7], r[8], round(punteggio, 4))


def cerca(indice: Indice, query: str, filtri: Filtri = Filtri(), k: int = 8) -> list[Risultato]:
    cond, par = _condizioni(filtri)
    ammessi = None
    if cond:
        sql = f"SELECT c.id FROM chunk c JOIN file f ON f.percorso = c.percorso WHERE {cond}"
        ammessi = {r[0] for r in indice.db.execute(sql, par)}
        if not ammessi:
            return []
    punteggi: dict[int, float] = {}
    for classifica in (_vettoriale(indice, query, ammessi), _bm25(indice, query, cond, par)):
        for rank, cid in enumerate(classifica):
            punteggi[cid] = punteggi.get(cid, 0.0) + 1 / (RRF_K + rank + 1)
    migliori = sorted(punteggi, key=punteggi.__getitem__, reverse=True)[:k]
    return [_carica(indice, cid, punteggi[cid]) for cid in migliori]


def simili(indice: Indice, nome: str, testo: str, k: int = 5) -> list[Simile]:
    _, chunks = chunk_di(indice.config, nome, testo)
    m = indice.matrice()
    if not chunks or not len(m.ids):
        return []
    s = indice.embedder.embed([c.testo_embed for c in chunks]) @ m.vettori.T
    inizi = np.flatnonzero(np.r_[True, m.percorsi[1:] != m.percorsi[:-1]])
    blocchi = np.maximum.reduceat(s, inizi, axis=1)  # (chunk nuovi, gruppi di file)
    per_file: dict[str, np.ndarray] = {}
    for j, i in enumerate(inizi):
        p = m.percorsi[i]
        per_file[p] = np.maximum(per_file[p], blocchi[:, j]) if p in per_file else blocchi[:, j]
    risultati = [Simile(p, round(float(b.max()), 4), round(float((b >= SOGLIA_COPERTURA).mean()), 4))
                 for p, b in per_file.items()]
    return sorted(risultati, key=lambda x: x.similarita, reverse=True)[:k]


def formatta_risultati(risultati: list[Risultato]) -> str:
    blocchi = []
    for i, r in enumerate(risultati, 1):
        info = [r.tipo, r.progetto, r.data, " + ".join(r.speaker) if r.speaker else None]
        dettagli = "; ".join(x for x in info if x)
        blocchi.append(f"[{i}] {r.citazione}  (righe {r.riga_inizio}-{r.riga_fine}; {dettagli})\n{r.testo}")
    return "\n\n---\n\n".join(blocchi)
```

"""Wrapper minimi su git via subprocess."""
from __future__ import annotations

import os
import subprocess
from dataclasses import dataclass
from pathlib import Path

_ENV = {"GIT_LITERAL_PATHSPECS": "1", "GIT_TERMINAL_PROMPT": "0"}


class GitError(RuntimeError):
    pass


def _esegui(root: Path, args: tuple[str, ...], timeout: float | None) -> bytes:
    try:
        r = subprocess.run(["git", "-C", str(root), *args], capture_output=True,
                           timeout=timeout, env={**os.environ, **_ENV})
    except subprocess.TimeoutExpired:
        raise GitError(f"git {args[0]}: timeout dopo {timeout}s") from None
    if r.returncode != 0:
        raise GitError(f"git {' '.join(args[:2])}: {r.stderr.decode(errors='replace').strip()}")
    return r.stdout


def git(root: Path, *args: str, timeout: float | None = None) -> str:
    return _esegui(root, args, timeout).decode("utf-8", errors="replace")


def git_bytes(root: Path, *args: str, timeout: float | None = None) -> bytes:
    return _esegui(root, args, timeout)


def lista_z(uscita: str) -> list[str]:
    return [x for x in uscita.split("\0") if x]


def in_repo(root: Path) -> bool:
    try:
        git(root, "rev-parse", "--is-inside-work-tree")
        return True
    except GitError:
        return False


def upstream(root: Path) -> str | None:
    try:
        return git(root, "rev-parse", "--abbrev-ref", "--symbolic-full-name", "@{u}").strip()
    except GitError:
        return None


def fetch(root: Path, timeout: float = 3.0) -> str | None:
    try:
        git(root, "fetch", "--quiet", timeout=timeout)
        return None
    except GitError as e:
        return str(e)


def in_staging(root: Path) -> list[str]:
    return lista_z(git(root, "diff", "--cached", "--name-only", "-z", "--diff-filter=ACMR"))


def git_add(root: Path, percorsi: list[str]) -> None:
    for i in range(0, len(percorsi), 200):
        git(root, "add", "--", *percorsi[i:i + 200])


@dataclass(frozen=True)
class Commit:
    hash: str
    autore: str
    quando: str
    messaggio: str


def commit_che_toccano(root: Path, intervallo: str, percorso: str) -> list[Commit]:
    uscita = git(root, "log", intervallo, "--format=%h%x1f%an%x1f%ar%x1f%s", "--", percorso)
    return [Commit(*riga.split("\x1f", 3)) for riga in uscita.splitlines() if riga]


def remoti_cambiati(root: Path) -> list[str]:
    return lista_z(git(root, "diff", "--name-only", "-z", "HEAD...@{u}"))
```

```python
"""Guardia: un file che stai per committare/modificare è stato cambiato sul remoto?"""
from __future__ import annotations

from dataclasses import dataclass, field
from pathlib import Path

from nb.gitutil import Commit, commit_che_toccano, fetch, upstream


@dataclass(frozen=True)
class Obsoleto:
    percorso: str
    commit: tuple[Commit, ...]

    def messaggio(self) -> str:
        righe = [f"✋ {self.percorso} è stato modificato sul remoto"]
        righe += [f'   {c.hash} — {c.autore}, {c.quando}: "{c.messaggio}"' for c in self.commit]
        righe.append("   Fai: git pull --rebase   poi riprova")
        return "\n".join(righe)


@dataclass
class EsitoGuardia:
    obsoleti: list[Obsoleto] = field(default_factory=list)
    avvisi: list[str] = field(default_factory=list)


def controlla(root: Path, percorsi: list[str], fai_fetch: bool = True, timeout: float = 3.0) -> EsitoGuardia:
    esito = EsitoGuardia()
    if upstream(root) is None:
        esito.avvisi.append("branch senza upstream: controllo delle modifiche remote saltato")
        return esito
    if fai_fetch:
        errore = fetch(root, timeout)
        if errore:
            esito.avvisi.append(f"fetch non riuscito ({errore}): controllo sull'ultimo stato noto del remoto")
    for p in percorsi:
        commit = commit_che_toccano(root, "HEAD..@{u}", p)
        if commit:
            esito.obsoleti.append(Obsoleto(p, tuple(commit)))
    return esito
```

def richiesti_a(config: Config, rev: str, percorsi: list[str]) -> dict[str, tuple[str, str]]:
    """hash -> (percorso md, testo da embeddare). rev="" legge dallo staging."""
    out: dict[str, tuple[str, str]] = {}
    for p in percorsi:
        if not indicizzabile(p):
            continue
        try:
            testo = git_bytes(config.root, "show", f"{rev}:{p}").decode("utf-8", errors="replace")
        except GitError:
            continue  # file cancellato in quella revisione
        _, chunks = chunk_di(config, p, testo)
        for c in chunks:
            out[chiave(config.model_id, c.testo_embed)] = (p, c.testo_embed)
    return out


def emb_tracciati(root: Path, rev: str) -> set[str]:
    return set(lista_z(git(root, "ls-tree", "-r", "--name-only", "-z", rev, "--", ".nb/emb")))


def mancanti_a(config: Config, rev: str, percorsi: list[str]) -> dict[str, tuple[str, str]]:
    tracciati = emb_tracciati(config.root, rev)
    return {h: v for h, v in richiesti_a(config, rev, percorsi).items() if percorso_rel(h) not in tracciati}


def md_tracciati(root: Path, rev: str = "HEAD") -> list[str]:
    return [p for p in lista_z(git(root, "ls-tree", "-r", "--name-only", "-z", rev)) if indicizzabile(p)]

from __future__ import annotations

import json
import subprocess
import sys
from pathlib import Path

from nb.coerenza import mancanti_a, richiesti_a
from nb.config import carica
from nb.embedder import crea

from nb.vettori import assicura, percorso_rel

NOMI = ("pre-commit", "pre-push", "post-merge", "post-checkout", "post-rewrite")
ZERO = "0" * 40
SCRIPT = """#!/bin/sh
# generato da `nb init`: non modificare a mano
command -v nb >/dev/null 2>&1 || {{ echo "nb non installato: hook {nome} saltato (vedi README)" >&2; exit 0; }}
exec nb hook {nome} "$@"
"""


def _err(*righe: str) -> None:
    for r in righe:
        print(r, file=sys.stderr)


def installa(root: Path) -> None:
    cartella = root / ".githooks"
    cartella.mkdir(exist_ok=True)
    for nome in NOMI:
        p = cartella / nome
        p.write_text(SCRIPT.format(nome=nome), encoding="utf-8")
        p.chmod(0o755)
    git(root, "config", "core.hooksPath", ".githooks")


def _stampa_guardia(esito: EsitoGuardia) -> None:
    _err(*(f"⚠ {a}" for a in esito.avvisi))
    _err(*(o.messaggio() for o in esito.obsoleti))


def pre_commit(root: Path) -> int:
    config = carica(root)
    staged = in_staging(root)
    esiti = []
    for p in staged:
        esiti += controlla_file(p, git_bytes(root, "show", f":{p}"), config.max_file_mb)
    da_guardare = [p for p in staged if not p.startswith(".nb/")]
    guardia = controlla(root, da_guardare) if da_guardare else EsitoGuardia()
    _err(*(str(e) for e in esiti))
    _stampa_guardia(guardia)
    if bloccanti(esiti) or guardia.obsoleti:
        _err("commit bloccato da nb (solo in emergenza: git commit --no-verify)")
        return 1
    richiesti = richiesti_a(config, "", [p for p in staged if indicizzabile(p)])
    if richiesti:
        assicura(config.emb_dir, crea(config), {h: t for h, (_, t) in richiesti.items()})
        git_add(root, [percorso_rel(h) for h in richiesti])
    return 0


def _file_da_pushare(root: Path, locale: str, remoto: str) -> list[str]:
    if remoto == ZERO:
        return lista_z(git(root, "log", "--name-only", "-z", "--format=", locale, "--not", "--remotes"))
    return lista_z(git(root, "diff", "--name-only", "-z", "--diff-filter=ACMR", remoto, locale))


def pre_push(root: Path, stdin: str) -> int:
    config = carica(root)
    toccati: set[str] = set()
    senza_vettori: set[str] = set()
    for riga in stdin.splitlines():
        parti = riga.split()
        if len(parti) != 4 or parti[1] == ZERO:
            continue  # riga malformata o cancellazione di branch
        _, locale, _, remoto = parti
        file = [p for p in _file_da_pushare(root, locale, remoto) if not p.startswith(".nb/")]
        toccati.update(file)
        senza_vettori.update(p for p, _ in mancanti_a(config, locale, file).values())
    guardia = controlla(root, sorted(toccati)) if toccati else EsitoGuardia()
    _stampa_guardia(guardia)
    for p in sorted(senza_vettori):
        _err(f"✋ vettori mancanti per {p}: esegui `nb sync` (o ricommitta con gli hook attivi)")
    if guardia.obsoleti or senza_vettori:
        _err("push bloccato da nb")
        return 1
    return 0


def _riepilogo(uscita: str) -> str:
    voci = lista_z(uscita)
    conteggi = {"A": 0, "M": 0, "D": 0, "R": 0}
    i = 0
    while i < len(voci):
        stato = voci[i][0]
        passo = 3 if stato in "RC" else 2
        percorso = voci[i + passo - 1] if i + passo - 1 < len(voci) else ""
        if indicizzabile(percorso):
            conteggi[stato if stato in conteggi else "M"] += 1
        i += passo
    if not any(conteggi.values()):
        return ""
    return (f"nb: fonti {conteggi['A']} nuove, {conteggi['M']} modificate, {conteggi['D']} rimosse, "
            f"{conteggi['R']} spostate — indice in aggiornamento")


def _indicizza_in_background(root: Path) -> None:
    subprocess.Popen([sys.executable, "-m", "nb", "index", "--silenzioso"], cwd=root,
                     start_new_session=True, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)


def post_sync(root: Path, nome: str, args: list[str]) -> int:
    if nome == "post-checkout" and (len(args) < 3 or args[2] != "1"):
        return 0  # checkout di singoli file, non di branch
    prima = args[0] if nome == "post-checkout" else "ORIG_HEAD"
    try:
        riepilogo = _riepilogo(git(root, "diff", "--name-status", "-z", prima, "HEAD"))
    except GitError:
        riepilogo = ""
    if riepilogo:
        _err(riepilogo)
    _indicizza_in_background(root)
    return 0


def claude_pre_edit(root: Path, stdin: str) -> int:
    try:
        file_path = json.loads(stdin or "{}").get("tool_input", {}).get("file_path")
    except json.JSONDecodeError:
        return 0
    if not file_path:
        return 0
    try:
        rel = Path(file_path).resolve().relative_to(root.resolve()).as_posix()
    except ValueError:
        return 0
    if not indicizzabile(rel):
        return 0
    esito = controlla(root, [rel])
    if esito.obsoleti:
        _err(esito.obsoleti[0].messaggio(),
             "Prima di modificarlo aggiorna il repo con `git pull --rebase` (chiedi conferma all'utente).")
        return 2
    return 0

"""Caricamento guidato: proposta di collocazione (contesto per Claude) ed esecuzione del piano."""
from __future__ import annotations

import json
from dataclasses import asdict, dataclass, field
from pathlib import Path, PurePosixPath

import yaml

from nb.chunker import RE_TITOLO, chunk_di
from nb.config import CARTELLA_TIPO, Config
from nb.fonti import indicizzabile, separa_frontmatter
from nb.gitutil import GitError, git, git_add, in_staging, lista_z, upstream
from nb.guardia import controlla
from nb.indice import Indice
from nb.regole import bloccanti, controlla_md, controlla_percorso
from nb.ricerca import simili
from nb.vettori import chiave, percorso_rel

SOGLIA_DUPLICATO = 0.8
CONVENZIONI = (
    "Cartelle: progetti/<progetto>/analisi|documentazione|call/ oppure generale/. "
    "Nomi: minuscolo-con-trattini.md; trascrizioni AAAA-MM-GG-titolo.md dentro call/. "
    "Frontmatter: tipo (analisi|documentazione|trascrizione|altro), progetto, data AAAA-MM-GG, "
    "partecipanti (solo trascrizioni). Preferisci la cartella dei documenti più simili; "
    "probabile duplicato → proponi aggiorna=true sul file esistente; nuovo progetto solo con conferma esplicita."
)


class CaricamentoError(RuntimeError):
    pass


@dataclass(frozen=True)
class VocePiano:
    sorgente: str
    destinazione: str
    frontmatter: dict = field(default_factory=dict)
    aggiorna: bool = False


def componi(originale: str, frontmatter: dict) -> str:
    fm, corpo, _, _ = separa_frontmatter(originale)
    unito = {k: v for k, v in {**(fm or {}), **frontmatter}.items() if v not in (None, "", [])}
    corpo = corpo.lstrip("\n")
    if not corpo.endswith("\n"):
        corpo += "\n"
    if not unito:
        return corpo
    return "---\n" + yaml.safe_dump(unito, allow_unicode=True, sort_keys=False) + "---\n" + corpo


def _titolo(corpo: str) -> str | None:
    for riga in corpo.split("\n"):
        m = RE_TITOLO.match(riga)
        if m:
            return m.group(2)
    return None


def proponi(indice: Indice, sorgenti: list[str]) -> dict:
    indice.sync()
    cartella_progetti = indice.config.root / "progetti"
    progetti = sorted(p.name for p in cartella_progetti.iterdir() if p.is_dir()) if cartella_progetti.is_dir() else []
    proposte = []
    for s in sorgenti:
        percorso = Path(s).expanduser()
        try:
            testo = percorso.read_text(encoding="utf-8", errors="replace")
        except OSError as e:
            proposte.append({"sorgente": s, "errore": f"illeggibile: {e}"})
            continue
        fm, corpo, _, _ = separa_frontmatter(testo)
        sim = simili(indice, percorso.name, corpo)
        proposte.append({
            "sorgente": s,
            "titolo": _titolo(corpo) or percorso.stem,
            "parole": len(corpo.split()),
            "frontmatter_presente": json.loads(json.dumps(fm or {}, default=str)),
            "simili": [asdict(x) for x in sim],
            "probabile_duplicato": next((x.percorso for x in sim if x.copertura >= SOGLIA_DUPLICATO), None),
        })
    return {"proposte": proposte, "progetti": progetti, "cartelle": list(CARTELLA_TIPO), "convenzioni": CONVENZIONI}


def _vettori_richiesti(config: Config, contenuti: dict[str, str]) -> list[str]:
    rel = set()
    for dest, testo in contenuti.items():
        _, chunks = chunk_di(config, dest, testo)
        rel.update(percorso_rel(chiave(config.model_id, c.testo_embed)) for c in chunks)
    return sorted(rel)


def _messaggio(contenuti: dict[str, str]) -> str:
    if len(contenuti) == 1:
        return f"nb: carica {next(iter(contenuti))}"
    return f"nb: carica {len(contenuti)} fonti"


def _pota(root: Path, p: Path) -> None:
    d = p.parent
    while d != root and d.is_dir() and not any(d.iterdir()):
        d.rmdir()
        d = d.parent


def _ripristina(indice: Indice, base: str, contenuti: dict[str, str], creati: list[str],
                vettori: list[str], nuovi: list[Path]) -> None:
    root = indice.config.root
    try:
        git(root, "reset", "--quiet")  # lo staging era vuoto prima di iniziare
    except GitError:
        pass
    for d in contenuti:
        if d in creati:
            (root / d).unlink(missing_ok=True)
            _pota(root, root / d)
        else:
            git(root, "restore", "--source", base, "--worktree", "--", d)
    nuovi_set = set(nuovi)
    for rel in vettori:
        p = root / rel
        if p in nuovi_set:
            p.unlink(missing_ok=True)
    indice.sync()


def _valida(indice: Indice, piano: list[VocePiano]) -> tuple[dict[str, str], dict[str, bool], list]:
    contenuti: dict[str, str] = {}
    aggiorna: dict[str, bool] = {}
    esiti = []
    for v in piano:
        dest = v.destinazione.strip()
        if dest.startswith("/") or ".." in PurePosixPath(dest).parts or not indicizzabile(dest):
            raise CaricamentoError(f"destinazione non valida: {dest!r} (percorso relativo .md, fuori da cartelle nascoste)")
        if dest in contenuti:
            raise CaricamentoError(f"destinazione ripetuta nel piano: {dest}")
        try:
            originale = Path(v.sorgente).expanduser().read_text(encoding="utf-8", errors="replace")
        except OSError as e:
            raise CaricamentoError(f"sorgente illeggibile: {v.sorgente} ({e})") from None
        contenuti[dest] = componi(originale, v.frontmatter)
        aggiorna[dest] = v.aggiorna
        esiti += controlla_percorso(dest) + controlla_md(dest, contenuti[dest])
    blocchi = bloccanti(esiti)
    if blocchi:
        raise CaricamentoError("\n".join(map(str, blocchi)))
    return contenuti, aggiorna, esiti


def carica_fonti(indice: Indice, piano: list[VocePiano], messaggio: str | None = None, push: bool = True) -> dict:
    root = indice.config.root
    if not piano:
        raise CaricamentoError("piano vuoto")
    contenuti, aggiorna, esiti = _valida(indice, piano)
    if in_staging(root):
        raise CaricamentoError("ci sono già file in staging: committali o toglili (git restore --staged) prima di caricare")
    sporchi = lista_z(git(root, "status", "--porcelain", "-z", "--", *contenuti))
    if sporchi:
        raise CaricamentoError("modifiche locali non committate su: " + ", ".join(s[3:] for s in sporchi))
    up = upstream(root)
    if up:
        git(root, "pull", "--rebase", "--autostash", "--quiet")
    base = git(root, "rev-parse", "HEAD").strip()
    for dest in contenuti:
        esiste = (root / dest).exists()
        if esiste and not aggiorna[dest]:
            raise CaricamentoError(f"{dest} esiste già: usa aggiorna=true oppure un altro nome")
        if aggiorna[dest] and not esiste:
            raise CaricamentoError(f"{dest} non esiste: niente da aggiornare")
    creati = [d for d in contenuti if not (root / d).exists()]
    vettori = _vettori_richiesti(indice.config, contenuti)
    nuovi: list[Path] = []
    try:
        for dest, testo in contenuti.items():
            (root / dest).parent.mkdir(parents=True, exist_ok=True)
            (root / dest).write_text(testo, encoding="utf-8")
        nuovi = indice.sync().vettori_nuovi
        obsoleti = controlla(root, list(contenuti), fai_fetch=False).obsoleti
        if obsoleti:
            raise CaricamentoError("\n".join(o.messaggio() for o in obsoleti))
        git_add(root, [*contenuti, *vettori])
        git(root, "commit", "--no-verify", "--quiet", "-m", messaggio or _messaggio(contenuti))
    except Exception:
        _ripristina(indice, base, contenuti, creati, vettori, nuovi)
        raise
    commit = git(root, "rev-parse", "--short", "HEAD").strip()
    pushato = False
    if push and up:
        try:
            git(root, "push", "--quiet")
        except GitError as e:
            git(root, "reset", "--quiet", "--mixed", base)
            _ripristina(indice, base, contenuti, creati, vettori, nuovi)
            raise CaricamentoError(f"push rifiutato, caricamento annullato: {e}") from None
        pushato = True
    avvisi = [str(e) for e in esiti]
    if not up:
        avvisi.append("branch senza upstream: commit solo locale, nessun push")
    return {"commit": commit, "pushato": pushato, "file": list(contenuti), "avvisi": avvisi}
```

```python
"""Operazioni di alto livello condivise da CLI e server MCP."""
from __future__ import annotations

import json

from nb.chunker import RE_FENCE, RE_TITOLO
from nb.coerenza import mancanti_a, md_tracciati
from nb.config import Config
from nb.fonti import indicizzabile
from nb.gitutil import fetch, git, git_add, in_repo, lista_z, remoti_cambiati, upstream
from nb.indice import Indice
from nb.vettori import assicura, percorso_rel


def stato(indice: Indice) -> dict:
    rep = indice.sync()
    root = indice.config.root
    db = indice.db
    risultato = {
        "file": db.execute("SELECT COUNT(*) FROM file").fetchone()[0],
        "chunk": db.execute("SELECT COUNT(*) FROM chunk").fetchone()[0],
        "modello": indice.config.model_id,
        "appena_sincronizzati": {"aggiunti": rep.aggiunti, "modificati": rep.modificati, "rimossi": rep.rimossi},
        "upstream": None,
        "vettori_non_committati": 0,
        "modificati_sul_remoto": [],
        "avvisi": indice.avvisi(),
    }
    if in_repo(root):
        risultato["vettori_non_committati"] = len(lista_z(
            git(root, "ls-files", "--others", "--exclude-standard", "-z", "--", ".nb/emb")))
        risultato["upstream"] = upstream(root)
        if risultato["upstream"]:
            errore = fetch(root)
            if errore:
                risultato["avvisi"]["(git)"] = [f"fetch non riuscito: {errore}"]
            risultato["modificati_sul_remoto"] = remoti_cambiati(root)
    return risultato


def elenca(indice: Indice, progetto: str | None = None, tipo: str | None = None) -> list[dict]:
    indice.sync()
    cond, par = [], []
    if progetto:
        cond.append("f.progetto = ?")
        par.append(progetto)
    if tipo:
        cond.append("f.tipo = ?")
        par.append(tipo)
    sql = ("SELECT f.percorso, f.tipo, f.progetto, f.data, f.partecipanti, COUNT(c.id) "
           "FROM file f LEFT JOIN chunk c ON c.percorso = f.percorso"
           + (" WHERE " + " AND ".join(cond) if cond else "")
           + " GROUP BY f.percorso ORDER BY f.percorso")
    return [{"percorso": p, "tipo": t, "progetto": pr, "data": d, "partecipanti": json.loads(pa), "chunk": n}
            for p, t, pr, d, pa, n in indice.db.execute(sql, par)]


def leggi(config: Config, percorso: str, sezione: str | None = None) -> str:
    root = config.root
    p = (root / percorso).resolve()
    if not p.is_relative_to(root) or not indicizzabile(percorso) or not p.is_file():
        raise ValueError(f"fonte non trovata: {percorso}")
    testo = p.read_text(encoding="utf-8", errors="replace")
    if not sezione:
        return testo
    cercato = sezione.split("›")[-1].strip().lower()
    righe = testo.split("\n")
    inizio, livello, titoli, in_codice = None, 0, [], False
    for i, riga in enumerate(righe):
        if RE_FENCE.match(riga):
            in_codice = not in_codice
            continue
        m = None if in_codice else RE_TITOLO.match(riga)
        if not m:
            continue
        titoli.append(m.group(2))
        if inizio is None and m.group(2).strip().lower() == cercato:
            inizio, livello = i, len(m.group(1))
        elif inizio is not None and len(m.group(1)) <= livello:
            return "\n".join(righe[inizio:i]).rstrip() + "\n"
    if inizio is None:
        raise ValueError(f"sezione '{sezione}' non trovata in {percorso}. Sezioni: {', '.join(titoli)}")
    return "\n".join(righe[inizio:]).rstrip() + "\n"


def sincronizza(indice: Indice) -> dict:
    """git pull + indice + commit/push dei vettori mancanti per i .md già committati."""
    config = indice.config
    root = config.root
    up = upstream(root)
    if up:
        git(root, "pull", "--rebase", "--autostash", "--quiet")
    rep = indice.sync()
    mancanti = mancanti_a(config, "HEAD", md_tracciati(root))
    pushato = False
    if mancanti:
        assicura(config.emb_dir, indice.embedder, {h: t for h, (_, t) in mancanti.items()})
        git_add(root, [percorso_rel(h) for h in mancanti])
        git(root, "commit", "--no-verify", "--quiet", "-m", f"nb: {len(mancanti)} vettori mancanti",
            "--only", "--", ".nb/emb")
        if up:
            git(root, "push", "--quiet")
            pushato = True
    return {"aggiunti": rep.aggiunti, "modificati": rep.modificati, "rimossi": rep.rimossi,
            "vettori_committati": len(mancanti), "pushato": pushato}
```

```python
"""`nb init`: prepara un knowledge repo (config, gitignore, hook, template per Claude Code)."""
from __future__ import annotations

from importlib.resources import files
from pathlib import Path

from nb.config import scrivi_default
from nb.gitutil import git, in_repo
from nb.hooks import installa

RIGA_GITIGNORE = ".nb/index.db*"
MODELLI = {
    "README.md": "README.md",
    "CLAUDE.md": "CLAUDE.md",
    ".claude/skills/nb-carica/SKILL.md": "skill-nb-carica.md",
    ".claude/settings.json": "settings.json",
    ".mcp.json": "mcp.json",
}


class InitError(RuntimeError):
    pass


def _gitignore(root: Path) -> bool:
    p = root / ".gitignore"
    testo = p.read_text(encoding="utf-8") if p.exists() else ""
    if RIGA_GITIGNORE in testo.splitlines():
        return False
    p.write_text(testo + ("" if not testo or testo.endswith("\n") else "\n") + RIGA_GITIGNORE + "\n", encoding="utf-8")
    return True


def inizializza(dove: Path) -> list[str]:
    if not in_repo(dove):
        raise InitError("non è un repo git: clona il knowledge repo (o esegui `git init`) e riprova")
    root = Path(git(dove, "rev-parse", "--show-toplevel").strip()).resolve()
    messaggi = []
    if scrivi_default(root):
        messaggi.append("creato .nb/config.toml")
    if _gitignore(root):
        messaggi.append(f"aggiunto {RIGA_GITIGNORE} a .gitignore")
    installa(root)
    messaggi.append("hook git attivi (.githooks, core.hooksPath)")
    for dest, modello in MODELLI.items():
        p = root / dest
        if p.exists():
            messaggi.append(f"{dest} esiste già: non toccato")
            continue
        p.parent.mkdir(parents=True, exist_ok=True)
        p.write_text(files("nb.modelli").joinpath(modello).read_text(encoding="utf-8"), encoding="utf-8")
        messaggi.append(f"creato {dest}")
    messaggi.append("Se è il primo init del repo: git add -A && git commit -m 'nb init' && git push. "
                    "Ogni collega, dopo il clone, esegue solo `nb init`.")
    return messaggi
```

from __future__ import annotations

import json
import shutil
import sys

import typer

from nb import hooks, servizi

app = typer.Typer(help="nb: knowledge base del team su git, interrogata da Claude Code.",
                  no_args_is_help=True, add_completion=False)


def _esci(messaggio: str, codice: int) -> None:
    typer.echo(messaggio, err=True)
    raise typer.Exit(codice)


def _indice() -> Indice:
    try:
        config = carica(trova_root(Path.cwd()))
    except ConfigError as e:
        _esci(str(e), 2)
    return Indice(config, crea(config))


def _json(dati) -> None:
    typer.echo(json.dumps(dati, ensure_ascii=False, indent=1, default=str))


@app.command()
def init() -> None:
    """Prepara il knowledge repo corrente (config, hook, template per Claude Code)."""
    try:
        for messaggio in inizializza(Path.cwd()):
            typer.echo(messaggio)
    except InitError as e:
        _esci(str(e), 2)


@app.command()
def index(silenzioso: bool = typer.Option(False, "--silenzioso")) -> None:
    """Aggiorna l'indice locale (solo i file cambiati)."""
    rep = _indice().sync()
    if not silenzioso:
        typer.echo(f"aggiunti {len(rep.aggiunti)}, modificati {len(rep.modificati)}, "
                   f"rimossi {len(rep.rimossi)}, vettori calcolati {len(rep.vettori_nuovi)}")


@app.command()
def search(query: str,
           progetto: str | None = typer.Option(None),
           tipo: str | None = typer.Option(None),
           da: str | None = typer.Option(None, help="data minima AAAA-MM-GG"),
           a: str | None = typer.Option(None, help="data massima AAAA-MM-GG"),
           k: int = typer.Option(8, "-k")) -> None:
    """Cerca nelle fonti."""
    indice = _indice()
    indice.sync()
    risultati = cerca(indice, query, Filtri(progetto, tipo, da, a), k)
    typer.echo(formatta_risultati(risultati) if risultati else "nessun risultato")


@app.command()
def status() -> None:
    """Stato di indice, vettori e remoto."""
    _json(servizi.stato(_indice()))


@app.command()
def check(file: Path) -> None:
    """Controlla se un file è stato modificato sul remoto (exit 1 se sì)."""
    root = _indice().config.root
    try:
        rel = file.resolve().relative_to(root).as_posix()
    except ValueError:
        _esci(f"{file} non è dentro il knowledge repo", 2)
    esito = controlla(root, [rel])
    for a in esito.avvisi:
        typer.echo(f"⚠ {a}")
    if esito.obsoleti:
        _esci("\n".join(o.messaggio() for o in esito.obsoleti), 1)
    typer.echo(f"✓ {rel} aggiornato")


@app.command()
def sync() -> None:
    """git pull + indice + commit/push dei vettori mancanti."""
    try:
        _json(servizi.sincronizza(_indice()))
    except GitError as e:
        _esci(str(e), 1)


@app.command()
def gc() -> None:
    """Rimuove i vettori non più usati da nessun .md presente."""
    indice = _indice()
    indice.sync()
    rimossi = gc_vettori(indice.config.emb_dir, indice.hash_vivi())
    typer.echo(f"rimossi {len(rimossi)} vettori orfani" + (" (ricordati di committare .nb/)" if rimossi else ""))


@app.command()
def reembed(si: bool = typer.Option(False, "--si", help="non chiedere conferma")) -> None:
    """Ricalcola tutti i vettori (dopo un cambio di modello in .nb/config.toml)."""
    indice = _indice()
    config = indice.config
    if not si:
        typer.confirm(f"Cancello .nb/emb e ricalcolo tutti i vettori con {config.model_id}. Procedo?", abort=True)
    indice.close()
    shutil.rmtree(config.emb_dir, ignore_errors=True)
    for suffisso in ("", "-wal", "-shm"):
        Path(f"{config.index_path}{suffisso}").unlink(missing_ok=True)
    rep = Indice(config, crea(config)).sync()
    typer.echo(f"{len(rep.vettori_nuovi)} vettori ricalcolati. Ora: git add -A .nb && "
               "git commit -m 'nb: reembed' && git push")


@app.command(context_settings={"allow_extra_args": True, "ignore_unknown_options": True})
def hook(ctx: typer.Context, nome: str) -> None:
    """Entry point degli hook (usato da .githooks e da Claude Code)."""
    try:
        root = trova_root(Path.cwd())
    except ConfigError:
        raise typer.Exit(0)  # non è un knowledge repo: niente da controllare
    if nome == "pre-commit":
        codice = hooks.pre_commit(root)
    elif nome == "pre-push":
        codice = hooks.pre_push(root, sys.stdin.read())
    elif nome in ("post-merge", "post-checkout", "post-rewrite"):
        codice = hooks.post_sync(root, nome, list(ctx.args))
    elif nome == "claude-pre-edit":
        codice = hooks.claude_pre_edit(root, sys.stdin.read())
    else:
        typer.echo(f"hook sconosciuto: {nome}", err=True)
        codice = 2
    raise typer.Exit(codice)


@app.command()
def mcp() -> None:
    """Avvia il server MCP su stdio (lanciato da Claude Code via .mcp.json)."""
    from nb.server import serve

    try:
        serve(trova_root(Path.cwd()))
    except ConfigError as e:
        _esci(str(e), 2)

```python
"""Server MCP: adattatore sottile per Claude Code."""
from __future__ import annotations

import json
from pathlib import Path

from mcp.server.fastmcp import FastMCP

from nb import caricamento, ricerca, servizi
from nb.config import carica
from nb.embedder import crea
from nb.indice import Indice

ISTRUZIONI = (
    "Motore di ricerca della knowledge base del team (analisi, documentazione, trascrizioni di call). "
    "Usa `cerca` prima di rispondere e cita sempre `percorso › sezione`; se le fonti non contengono "
    "la risposta dillo. Per caricare file: `proponi_collocazione`, mostra il piano, attendi conferma "
    "esplicita, poi `carica_fonti`."
)


def _json(dati) -> str:
    return json.dumps(dati, ensure_ascii=False, indent=1, default=str)


def crea_server(root: Path) -> FastMCP:
    config = carica(root)
    indice = Indice(config, crea(config))
    indice.sync()
    server = FastMCP("nb", instructions=ISTRUZIONI)

    @server.tool()
    def cerca(query: str, progetto: str | None = None, tipo: str | None = None,
              da: str | None = None, a: str | None = None, k: int = 8) -> str:
        """Cerca nelle fonti del team (semantica + parole chiave). Filtri facoltativi: progetto,
        tipo (analisi | documentazione | trascrizione | altro), da/a (date AAAA-MM-GG, incluse).
        Ogni risultato inizia con la citazione `percorso › sezione` da riportare nella risposta."""
        indice.sync()
        risultati = ricerca.cerca(indice, query, ricerca.Filtri(progetto, tipo, da, a), k)
        return ricerca.formatta_risultati(risultati) if risultati else "nessun risultato nelle fonti"

    @server.tool()
    def leggi_fonte(percorso: str, sezione: str | None = None) -> str:
        """Testo completo di una fonte, o di una sola sezione (titolo, anche nel formato `A › B`)."""
        return servizi.leggi(config, percorso, sezione)

    @server.tool()
    def elenca_fonti(progetto: str | None = None, tipo: str | None = None) -> str:
        """Elenco delle fonti con tipo, progetto, data, partecipanti e numero di chunk."""
        return _json(servizi.elenca(indice, progetto, tipo))

    @server.tool()
    def stato() -> str:
        """Stato dell'indice, vettori non committati, file modificati sul remoto e avvisi sui frontmatter."""
        return _json(servizi.stato(indice))

    @server.tool()
    def proponi_collocazione(file: list[str]) -> str:
        """Contesto per decidere dove caricare dei .md locali: documenti simili con la loro cartella,
        probabili duplicati, progetti esistenti e convenzioni. Non scrive nulla."""
        return _json(caricamento.proponi(indice, file))

    @server.tool()
    def carica_fonti(piano: list[dict], messaggio: str | None = None, push: bool = True) -> str:
        """Scrive i file del piano nel repo, calcola i vettori, fa UN commit e il push.
        Chiamalo SOLO dopo la conferma esplicita dell'utente sul piano.
        Ogni voce: {sorgente, destinazione, frontmatter, aggiorna}. In caso di errore annulla tutto."""
        voci = [caricamento.VocePiano(sorgente=v["sorgente"], destinazione=v["destinazione"],
                                      frontmatter=v.get("frontmatter") or {},
                                      aggiorna=bool(v.get("aggiorna", False))) for v in piano]
        return _json(caricamento.carica_fonti(indice, voci, messaggio, push))

    return server


def serve(root: Path) -> None:
    crea_server(root).run()
```
